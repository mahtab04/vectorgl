cmake_minimum_required(VERSION 3.20)
project(vectorgl_ui_consumer LANGUAGES CXX)
find_package(vectorgl CONFIG REQUIRED)
if(TARGET vectorgl::ui)
    message(FATAL_ERROR "Base package discovery imported the optional UI component")
endif()
find_package(vectorgl CONFIG REQUIRED COMPONENTS ui)
add_executable(vectorgl_ui_consumer main.cpp)
target_link_libraries(vectorgl_ui_consumer PRIVATE vectorgl::ui)
if(TARGET vectorgl::glfw)
    message(FATAL_ERROR "UI-only discovery imported the GLFW adapter")
endif()
if(TARGET vectorgl::dialogs)
    message(FATAL_ERROR "UI-only discovery imported native dialog backends")
endif()
option(VECTORGL_TEST_DIALOGS "Test the installed dialogs component" OFF)
if(VECTORGL_TEST_DIALOGS)
    find_package(vectorgl CONFIG REQUIRED COMPONENTS dialogs)
    target_link_libraries(vectorgl_ui_consumer PRIVATE vectorgl::dialogs)
    if(TARGET vectorgl::glfw)
        message(FATAL_ERROR "Dialogs discovery imported the GLFW adapter")
    endif()
endif()
option(VECTORGL_TEST_GLFW "Test the installed GLFW adapter component" OFF)
if(VECTORGL_TEST_GLFW)
    find_package(vectorgl CONFIG REQUIRED COMPONENTS glfw)
    target_link_libraries(vectorgl_ui_consumer PRIVATE vectorgl::glfw)
    target_compile_definitions(vectorgl_ui_consumer PRIVATE VECTORGL_TEST_GLFW=1)
endif()
enable_testing()
add_test(NAME vectorgl_ui_consumer COMMAND vectorgl_ui_consumer)

option(VECTORGL_TEST_WORKSPACE "Build the standalone workspace against the installed SDK" OFF)
option(VECTORGL_TEST_WORKSPACE_GPU "Run the installed workspace smoke test (needs a display)" OFF)
if(VECTORGL_TEST_WORKSPACE)
    if(NOT VECTORGL_TEST_GLFW)
        message(FATAL_ERROR "Installed workspace requires VECTORGL_TEST_GLFW")
    endif()
    add_executable(vectorgl_installed_workspace ../../example/ui_editor_workspace_demo.cpp)
    target_link_libraries(vectorgl_installed_workspace PRIVATE vectorgl::glfw)
    if(VECTORGL_TEST_DIALOGS)
        target_link_libraries(vectorgl_installed_workspace PRIVATE vectorgl::dialogs)
    endif()
    if(VECTORGL_TEST_WORKSPACE_GPU)
        set(workspace_runner)
        if(UNIX AND NOT APPLE)
            find_program(XVFB_RUN_EXECUTABLE NAMES xvfb-run)
            if(XVFB_RUN_EXECUTABLE)
                set(workspace_runner ${XVFB_RUN_EXECUTABLE} -a)
            endif()
        endif()
        add_test(NAME vectorgl_installed_workspace_smoke COMMAND ${workspace_runner}
            $<TARGET_FILE:vectorgl_installed_workspace> --smoke-test --width 1280 --height 780
            --font ${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/box.ttf)
        set_tests_properties(vectorgl_installed_workspace_smoke PROPERTIES TIMEOUT 45)
    endif()
endif()
