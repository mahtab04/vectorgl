cmake_minimum_required(VERSION 3.21)
project(vectorgl_embedding_test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(BUILD_SHARED_LIBS ON)
set(ENABLE_CLANG_TIDY ON) # A parent's generic option must not enable VectorGL tooling.
set(ENABLE_FORMAT_CHECK ON)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "Parent preference")
set(GLFW_INSTALL ON CACHE BOOL "Parent preference")
option(VECTORGL_EMBED_WITH_FETCHCONTENT "Exercise FetchContent rather than add_subdirectory" ON)
if(VECTORGL_EMBED_WITH_FETCHCONTENT)
    include(FetchContent)
    get_filename_component(vectorgl_source "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
    FetchContent_Declare(vectorgl SOURCE_DIR "${vectorgl_source}")
    FetchContent_MakeAvailable(vectorgl)
else()
    add_subdirectory(../.. vectorgl-build)
endif()
if(NOT CMAKE_CXX_STANDARD EQUAL 23 OR NOT BUILD_SHARED_LIBS OR GLFW_BUILD_EXAMPLES OR NOT GLFW_INSTALL)
    message(FATAL_ERROR "VectorGL changed the parent project's compiler/dependency settings")
endif()
foreach(option IN ITEMS BUILD_TESTS BUILD_APP_TESTS BUILD_APPS BUILD_EXAMPLES BUILD_UI_EXAMPLES
        BUILD_GPU_TESTS ENABLE_CLANG_TIDY ENABLE_FORMAT_CHECK)
    if(VECTORGL_${option})
        message(FATAL_ERROR "Embedded VectorGL unexpectedly enabled ${option}")
    endif()
endforeach()
foreach(module IN ITEMS vectorgl ui glfw dialogs)
    if(NOT TARGET vectorgl::${module})
        message(FATAL_ERROR "Default embedding must provide vectorgl::${module}")
    endif()
endforeach()
add_executable(vectorgl_embedded_consumer main.cpp)
target_link_libraries(vectorgl_embedded_consumer PRIVATE vectorgl::glfw vectorgl::dialogs)
get_target_property(glfw_type glfw TYPE)
if(WIN32 AND glfw_type STREQUAL "SHARED_LIBRARY")
    add_custom_command(TARGET vectorgl_embedded_consumer POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:glfw> $<TARGET_FILE_DIR:vectorgl_embedded_consumer>
        VERBATIM)
endif()
enable_testing()
add_test(NAME vectorgl_embedded_consumer COMMAND vectorgl_embedded_consumer)
set_tests_properties(vectorgl_embedded_consumer PROPERTIES LABELS package)
